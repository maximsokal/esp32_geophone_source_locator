# CS-IDENTITY-01: the build identity of the firmware. esp_app_desc.c (the "App version" / "Compile time" lines) is rebuilt
# only when CMake is configured again, so an incremental build keeps the old commit; this one is regenerated by every
# build from the tree (HEAD, and a hash of the uncommitted diff) and rewritten only when it changes.
idf_build_get_property(app_identity_python PYTHON)
set(APP_BUILD_IDENTITY_C "${CMAKE_CURRENT_BINARY_DIR}/app_build_identity.c")
add_custom_target(app_build_identity ALL
    COMMAND "${app_identity_python}" "${CMAKE_SOURCE_DIR}/tools/gen_build_identity.py"
            --repo "${CMAKE_SOURCE_DIR}" --out "${APP_BUILD_IDENTITY_C}"
    BYPRODUCTS "${APP_BUILD_IDENTITY_C}"
    COMMENT "Generating the firmware build identity")
set_source_files_properties("${APP_BUILD_IDENTITY_C}" PROPERTIES GENERATED TRUE)
target_sources(${COMPONENT_LIB} PRIVATE "${APP_BUILD_IDENTITY_C}")
add_dependencies(${COMPONENT_LIB} app_build_identity)
